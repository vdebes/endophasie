1"""Pause mode: show what you said each time you pause long enough (X11). 

2 

3Usage: streamer.py <rec.wav> <recorder pid> [lang] 

4 

5Started by `dictate` together with pw-record when DICTATE_PAUSE > 0. 

6Follows the WAV file as it grows. Short pauses (hesitations) do not cut: 

7only DICTATE_PAUSE seconds of silence send what was said since the last 

8cut to the server, with the text so far as context, and paste the result 

9into the window that had focus when recording started. 

10 

11Why long pauses only: cutting at every 300 ms pause (tried 2026-09-27) 

12split sentences in the middle, and Whisper closed each piece with "…". 

13 

14On SIGUSR1 (sent by dictate on stop), or if the recorder dies, it 

15transcribes what is left, pastes it, restores the clipboard and exits. 

16If focus moved to another window, text is held back instead of landing 

17there: pasted when focus comes back, else left in the clipboard at the end. 

18 

19Settings (environment or ~/.config/dictate/env): 

20 DICTATE_PAUSE silence (s) that shows the text; 0 (default) = off, 

21 one block after you stop (checked by dictate) 

22 DICTATE_AFTER_PAUSE continue (default): the mic stays open after the 

23 text is shown; stop: the mic closes 

24 DICTATE_MIC_IDLE continue mode: close the mic after this many 

25 seconds without speech (default: 30) 

26 DICTATE_DRYRUN=1 print chunks instead of pasting (tests) 

27"""

42SR = 16000 

43CHUNK_WAV = os.path.join(RUN, "chunk.wav") 

44REC_PID = os.path.join(RUN, "rec.pid") 

45NOTIF = os.path.join(RUN, "notify.id") 

46PAUSE_S = float(os.environ.get("DICTATE_PAUSE", "0")) 

47AFTER_PAUSE = os.environ.get("DICTATE_AFTER_PAUSE", "continue") 

48MIC_IDLE_S = float(os.environ.get("DICTATE_MIC_IDLE", "30")) 

49TAIL_S = PAUSE_S + 2 # VAD only looks at the end of the buffer: cheap even after minutes 

50PAD = SR * 3 // 10 # 0.3 s kept after the last word; long silences invite hallucinations 

51POLL_S = 0.25 

52DRYRUN = bool(os.environ.get("DICTATE_DRYRUN")) 

53TERMINALS = ("terminator", "terminal", "kitty", "alacritty", "konsole", "xterm", 

54 "tilix", "wezterm", "ghostty")

73def notify(msg: str, timeout_ms: int = 2000) -> None: 

74 """Replaces dictate's "recording" notification (same replace-id).""" 

75 try: 

76 old = open(NOTIF).read().strip() 

77 except FileNotFoundError:

79 nid = run(["notify-send", "-p", "-t", str(timeout_ms), *(["-r", old] if old else []),

87 # xclip forks a daemon that serves the selection and inherits our pipes: 

88 # capturing its output would block until someone else copies something.

93class WavTail: 

94 """Reads the samples pw-record appends to a WAV file that is still being 

95 written (the header's size fields are not final yet, so they are ignored).""" 

96 

97 def __init__(self, path: str): 

98 self.path, self.rest = path, b"" 

99 self.f: BinaryIO | None = None 

100 

101 def read(self) -> np.ndarray: 

102 if self.f is None: 

103 try: 

104 self.f = open(self.path, "rb") 

105 except FileNotFoundError: 

106 return np.zeros(0, np.float32) 

107 head = self.f.read(4096) 

108 i = head.find(b"data") 

109 if i < 0: # header not fully written yet 

110 self.f.close() 

111 self.f = None 

112 return np.zeros(0, np.float32) 

113 self.f.seek(i + 8) # "data" + 4-byte size 

114 raw = self.rest + self.f.read() 

115 cut = len(raw) - len(raw) % 2 # int16 samples: keep a stray byte 

116 self.rest = raw[cut:] 

117 return np.frombuffer(raw[:cut], np.int16).astype(np.float32) / 32768

120def speech_end(audio: np.ndarray) -> int | None: 

121 """Sample index where the last speech in `audio` ends, None if silent.""" 

122 if len(audio) == 0: 

123 return None 

124 sp = get_speech_timestamps( 

125 audio, VadOptions(min_silence_duration_ms=500, speech_pad_ms=100)) 

126 return sp[-1]["end"] if sp else None

129class Paster: 

130 """Pastes into the window that had focus at start, via the clipboard, 

131 and gives the clipboard its previous (text) content back at the end.""" 

132 

133 def __init__(self): 

134 self.win = run(["xdotool", "getactivewindow"]).strip() 

135 cls = run(["xprop", "-id", self.win, "WM_CLASS"]).lower() if self.win else "" 

136 self.key = "ctrl+shift+v" if any(t in cls for t in TERMINALS) else "ctrl+v" 

137 self.saved = None if DRYRUN else ( 

138 subprocess.run(["xclip", "-o", "-selection", "clipboard"], 

139 capture_output=True, text=True).stdout or None) 

140 self.pending: list[str] = [] 

141 self.pasted = False 

142 

143 def add(self, text: str) -> None: 

144 self.pending.append(text) 

145 self.flush() 

146 

147 def flush(self) -> None: 

148 if not self.pending: 

149 return 

150 text = " ".join(self.pending) 

151 if self.pasted: 

152 text = " " + text # space between chunks 

153 if DRYRUN: 

154 print(text, flush=True) 

155 else: 

156 if run(["xdotool", "getactivewindow"]).strip() != self.win: 

157 return # focus moved: hold back 

158 set_clipboard(text) 

159 run(["xdotool", "key", "--clearmodifiers", self.key]) 

160 time.sleep(0.3) # let the app read the clipboard 

161 self.pending.clear() 

162 self.pasted = True 

163 

164 def close(self) -> bool: 

165 """True if everything was pasted; else the held-back text is left 

166 in the clipboard.""" 

167 self.flush() 

168 if DRYRUN: 

169 return True 

170 if self.pending: 

171 set_clipboard(" ".join(self.pending)) 

172 return False 

173 if self.saved is not None: 

174 set_clipboard(self.saved) 

175 return True

178def close_mic(rec_pid: int, reason: str) -> None: 

179 """Stop the recording ourselves (no key press): SIGINT finalizes the WAV, 

180 and rec.pid goes away so the next shortcut starts a new recording 

181 instead of signalling a pid the system may have reused.""" 

182 try: 

183 os.kill(rec_pid, signal.SIGINT) 

184 except ProcessLookupError: 

185 pass 

186 try: 

187 if open(REC_PID).read().strip() == str(rec_pid): 

188 os.unlink(REC_PID) 

189 except FileNotFoundError: 

190 pass 

191 log(f"mic closed: {reason}")

194def main() -> int: 

195 wav_path, rec_pid = sys.argv[1], int(sys.argv[2]) 

196 lang = sys.argv[3] if len(sys.argv) > 3 else "" 

197 if PAUSE_S <= 0: 

198 return 0 

199 signal.signal(signal.SIGUSR1, on_stop) 

200 tail, paster = WavTail(wav_path), Paster() 

201 buf = np.zeros(0, np.float32) 

202 heard = False # buf holds speech not sent yet 

203 last_voice = time.monotonic() # for DICTATE_MIC_IDLE 

204 self_closed = "" # why we closed the mic, if we did 

205 words, chunks, audio_s, t_stop = 0, 0, 0.0, None 

206 context = "" 

207 

208 def emit(chunk: np.ndarray) -> None: 

209 nonlocal words, chunks, context 

210 with wave.open(CHUNK_WAV, "wb") as w: 

211 w.setnchannels(1) 

212 w.setsampwidth(2) 

213 w.setframerate(SR) 

214 w.writeframes((np.clip(chunk, -1, 1) * 32767).astype(np.int16).tobytes()) 

215 try: 

216 text = request(CHUNK_WAV, lang, context[-300:]).strip() 

217 except Exception as e: # keep going: next chunk may work 

218 log(f"chunk failed: {e}") 

219 return 

220 if text: 

221 paster.add(text) 

222 context = f"{context} {text}".strip() 

223 words += len(text.split()) 

224 chunks += 1 

225 

226 while True: 

227 new = tail.read() 

228 buf = np.concatenate([buf, new]) 

229 audio_s += len(new) / SR 

230 recorder_gone = not os.path.exists(f"/proc/{rec_pid}") 

231 if stopping or recorder_gone: 

232 t_stop = time.monotonic() 

233 buf = np.concatenate([buf, tail.read()]) # last bytes after SIGINT 

234 end = speech_end(buf) 

235 if end is not None: 

236 emit(buf[:end + PAD]) 

237 break 

238 now = time.monotonic() 

239 if len(buf) >= SR: 

240 start = max(0, len(buf) - int(TAIL_S * SR)) 

241 end = speech_end(buf[start:]) 

242 if end is not None: 

243 end += start 

244 heard = True 

245 last_voice = max(last_voice, now - (len(buf) - end) / SR) 

246 elif heard: 

247 # Silence longer than the window we look at: the loop was 

248 # busy (transcribing); find the end over the whole buffer. 

249 end = speech_end(buf) 

250 if heard and end is not None and len(buf) - end >= PAUSE_S * SR: 

251 if AFTER_PAUSE == "stop": # close first: nothing said during 

252 self_closed = "after the pause" # the transcription gets lost 

253 close_mic(rec_pid, self_closed) 

254 emit(buf[:end + PAD]) 

255 buf, heard = buf[-SR // 2:], False 

256 elif not heard: 

257 buf = buf[-SR // 2:] # silence only: drop it 

258 if (not self_closed and AFTER_PAUSE != "stop" 

259 and now - last_voice >= MIC_IDLE_S): 

260 self_closed = f"no speech for {MIC_IDLE_S:.0f} s" 

261 close_mic(rec_pid, self_closed) 

262 paster.flush() # focus may have come back 

263 time.sleep(POLL_S) 

264 

265 complete = paster.close() 

266 if os.path.exists(CHUNK_WAV): 

267 os.unlink(CHUNK_WAV) 

268 wait_ms = int((time.monotonic() - t_stop) * 1000) 

269 with open(os.path.join(RUN, "stats.log"), "a") as f: 

270 closed_by = "streamer" if self_closed else "key" 

271 print(f"{time.strftime('%T')} pause={PAUSE_S:g}s audio={audio_s:.0f}s chunks={chunks} " 

272 f"words={words} final_wait={wait_ms}ms closed_by={closed_by}", file=f) 

273 if not complete: 

274 notify("📋 Focus changed: text left in the clipboard", 4000) 

275 elif self_closed: 

276 # dictate's stop call did not run: its cleanup and "✅ Done" are ours. 

277 notify(f"⏹️ Mic closed ({self_closed})") 

278 if self_closed and tail.f is not None: 

279 # Delete the audio, unless a new recording already replaced the file. 

280 try: 

281 if os.stat(wav_path).st_ino == os.fstat(tail.f.fileno()).st_ino: 

282 os.unlink(wav_path) 

283 except FileNotFoundError: 

284 pass 

285 return 0 

286 

287 

288if __name__ == "__main__": 

289 sys.exit(main())
